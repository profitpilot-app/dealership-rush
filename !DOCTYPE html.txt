    <script>
        // Local Device Storage Controller Framework
        let playerWallet = {
            gc: parseInt(localStorage.getItem('slot_gc')) || 5000,
            sc: parseFloat(localStorage.getItem('slot_sc')) || 5.00
        };

        function syncAppWalletUI() {
            document.getElementById('app-gc').innerText = playerWallet.gc.toLocaleString();
            document.getElementById('app-sc').innerText = playerWallet.sc.toFixed(2);
            localStorage.setItem('slot_gc', playerWallet.gc);
            localStorage.setItem('slot_sc', playerWallet.sc);
        }

        function simulateAdNetworkReward() {
            alert("Simulating 30-Second Commercial Playback View...");
            playerWallet.gc += 1000;
            syncAppWalletUI();
        }

        // Initialize display parameters immediately on launch
        syncAppWalletUI();

        // PHASER MOBILE REEL CANVAS GRAPHICS 
        const phaserConfig = {
            type: Phaser.AUTO,
            width: 450,
            height: 380,
            parent: 'game-screen',
            backgroundColor: '#0d152d',
            scene: { preload: preload, create: create }
        };

        const appGameInstance = new Phaser.Game(phaserConfig);
        let screenReelSprites = [];

        function preload() {
            // High-quality public image assets for slot testing
            this.load.image('discRed', 'https://phaser.io');
            this.load.image('discYellow', 'https://phaser.io');
        }

        function create() {
            // Draw visual grid container background
            let boardBase = this.add.graphics();
            boardBase.fillStyle(0x18244a, 1);
            boardBase.fillRect(65, 40, 320, 240);

            let actionButton = this.add.text(140, 310, ' [ CLICK TO SPIN ] ', { fontSize: '20px', fill: '#4df0a6', fontStyle: 'bold', backgroundColor: '#060b1e' })
                .setPadding(8).setInteractive({ useHandCursor: true });

            actionButton.on('pointerdown', () => {
                actionButton.disableInteractive();
                actionButton.setAlpha(0.5);

                const currentMode = document.getElementById('app-currency-mode').value;
                const cost = currentMode === "GC" ? 100 : 0.50;

                // Local balance confirmation execution loop
                if (currentMode === "GC" && playerWallet.gc < cost) {
                    alert("Insufficient Gold Coins! Please watch an ad to top up.");
                    actionButton.setInteractive(); actionButton.setAlpha(1);
                    return;
                }
                if (currentMode === "SC" && playerWallet.sc < cost) {
                    alert("Insufficient Sweeps Entry Tokens.");
                    actionButton.setInteractive(); actionButton.setAlpha(1);
                    return;
                }

                // Process internal wager transactions inside the device memory
                if (currentMode === "GC") playerWallet.gc -= cost;
                if (currentMode === "SC") playerWallet.sc -= cost;

                // 35% casual hit ratio built directly into player browser client
                let localWinChance = Math.random() < 0.35; 
                let rewardEarnings = localWinChance ? cost * 3 : 0;

                if (currentMode === "GC") playerWallet.gc += rewardEarnings;
                if (currentMode === "SC") playerWallet.sc += rewardEarnings;

                // Play cascading bouncy dropdown reel animations
                animateLocalAppDrop(this, localWinChance);
                syncAppWalletUI();

                this.time.delayedCall(800, () => {
                    actionButton.setInteractive();
                    actionButton.setAlpha(1);
                });
            });
        }

        function animateLocalAppDrop(scene, forcedWinPattern) {
            screenReelSprites.forEach(s => s.destroy());
            screenReelSprites = [];

            for (let r = 0; r < 4; r++) {
                for (let c = 0; c < 5; c++) {
                    let coordinateX = 95 + (c * 65);
                    let coordinateY = 70 + (r * 55);

                    // If server math flags a win cycle, line up symbols horizontally on row 1
                    let resolveRed = forcedWinPattern ? (r === 1 ? true : Math.random() > 0.5) : Math.random() > 0.5;
                    let targetTex = resolveRed ? 'discRed' : 'discYellow';

                    let diskImage = scene.add.image(coordinateX, -40, targetTex).setScale(1.2);
                    screenReelSprites.push(diskImage);

                    scene.tweens.add({
                        targets: diskImage,
                        y: coordinateY,
                        duration: 400 + (c * 60),
                        ease: 'Bounce.easeOut'
                    });
                }
            }
        }
    </script>
